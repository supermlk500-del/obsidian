# 无序链表和有序链表的实现

我将分别实现无序链表和有序链表，并展示它们的查找、删除、大小等操作。

## 无序链表（Unordered Linked List）

无序链表中的元素没有特定的顺序，新元素通常添加到链表头部。

```python
class Node:
    """链表节点类"""
    def __init__(self, data):
        self.data = data
        self.next = None

class UnorderedLinkedList:
    """无序链表实现"""
    
    def __init__(self):
        self.head = None
    
    def is_empty(self):
        """检查链表是否为空"""
        return self.head is None
    
    def add(self, item):
        """在链表头部添加元素"""
        new_node = Node(item)
        new_node.next = self.head
        self.head = new_node
    
    def size(self):
        """返回链表大小"""
        current = self.head
        count = 0
        while current is not None:
            count += 1
            current = current.next
        return count
    
    def search(self, item):
        """查找元素是否存在"""
        current = self.head
        while current is not None:
            if current.data == item:
                return True
            current = current.next
        return False
    
    def remove(self, item):
        """删除指定元素"""
        current = self.head
        previous = None
        
        while current is not None:
            if current.data == item:
                break
            previous = current
            current = current.next
        
        # 如果没找到元素
        if current is None:
            raise ValueError(f"元素 {item} 不在链表中")
        
        # 如果删除的是头节点
        if previous is None:
            self.head = current.next
        else:
            previous.next = current.next
    
    def append(self, item):
        """在链表尾部添加元素"""
        new_node = Node(item)
        
        if self.is_empty():
            self.head = new_node
            return
        
        current = self.head
        while current.next is not None:
            current = current.next
        
        current.next = new_node
    
    def insert(self, pos, item):
        """在指定位置插入元素"""
        if pos <= 0:
            self.add(item)
            return
        
        if pos >= self.size():
            self.append(item)
            return
        
        new_node = Node(item)
        current = self.head
        count = 0
        
        while count < pos - 1:
            current = current.next
            count += 1
        
        new_node.next = current.next
        current.next = new_node
    
    def index(self, item):
        """返回元素的索引位置"""
        current = self.head
        index = 0
        
        while current is not None:
            if current.data == item:
                return index
            current = current.next
            index += 1
        
        raise ValueError(f"元素 {item} 不在链表中")
    
    def pop(self, pos=None):
        """删除并返回指定位置的元素，默认删除最后一个元素"""
        if self.is_empty():
            raise IndexError("链表为空")
        
        # 如果没有指定位置，删除最后一个元素
        if pos is None:
            pos = self.size() - 1
        
        # 检查位置是否有效
        if pos < 0 or pos >= self.size():
            raise IndexError("位置超出范围")
        
        # 删除头节点的情况
        if pos == 0:
            item = self.head.data
            self.head = self.head.next
            return item
        
        # 删除中间或尾部节点
        current = self.head
        previous = None
        count = 0
        
        while count < pos:
            previous = current
            current = current.next
            count += 1
        
        item = current.data
        previous.next = current.next
        return item
    
    def display(self):
        """显示链表内容"""
        elements = []
        current = self.head
        while current is not None:
            elements.append(str(current.data))
            current = current.next
        print(" -> ".join(elements) + " -> None")
```

## 有序链表（Ordered Linked List）

有序链表中的元素按照特定顺序（如升序）排列。

```python
class OrderedLinkedList:
    """有序链表实现（升序）"""
    
    def __init__(self):
        self.head = None
    
    def is_empty(self):
        """检查链表是否为空"""
        return self.head is None
    
    def size(self):
        """返回链表大小"""
        current = self.head
        count = 0
        while current is not None:
            count += 1
            current = current.next
        return count
    
    def add(self, item):
        """按顺序添加元素"""
        new_node = Node(item)
        
        # 如果链表为空或新元素小于头节点
        if self.is_empty() or item < self.head.data:
            new_node.next = self.head
            self.head = new_node
            return
        
        # 找到插入位置
        current = self.head
        previous = None
        
        while current is not None and current.data <= item:
            previous = current
            current = current.next
        
        # 插入新节点
        new_node.next = current
        previous.next = new_node
    
    def search(self, item):
        """查找元素是否存在（利用有序性优化）"""
        current = self.head
        
        while current is not None:
            if current.data == item:
                return True
            if current.data > item:
                # 由于链表有序，如果当前节点值大于目标值，说明目标值不存在
                return False
            current = current.next
        
        return False
    
    def remove(self, item):
        """删除指定元素"""
        current = self.head
        previous = None
        
        while current is not None:
            if current.data == item:
                break
            if current.data > item:
                # 由于链表有序，如果当前节点值大于目标值，说明目标值不存在
                raise ValueError(f"元素 {item} 不在链表中")
            previous = current
            current = current.next
        
        # 如果没找到元素
        if current is None:
            raise ValueError(f"元素 {item} 不在链表中")
        
        # 如果删除的是头节点
        if previous is None:
            self.head = current.next
        else:
            previous.next = current.next
    
    def index(self, item):
        """返回元素的索引位置（利用有序性优化）"""
        current = self.head
        index = 0
        
        while current is not None:
            if current.data == item:
                return index
            if current.data > item:
                # 由于链表有序，如果当前节点值大于目标值，说明目标值不存在
                raise ValueError(f"元素 {item} 不在链表中")
            current = current.next
            index += 1
        
        raise ValueError(f"元素 {item} 不在链表中")
    
    def pop(self, pos=None):
        """删除并返回指定位置的元素，默认删除最后一个元素"""
        if self.is_empty():
            raise IndexError("链表为空")
        
        # 如果没有指定位置，删除最后一个元素
        if pos is None:
            pos = self.size() - 1
        
        # 检查位置是否有效
        if pos < 0 or pos >= self.size():
            raise IndexError("位置超出范围")
        
        # 删除头节点的情况
        if pos == 0:
            item = self.head.data
            self.head = self.head.next
            return item
        
        # 删除中间或尾部节点
        current = self.head
        previous = None
        count = 0
        
        while count < pos:
            previous = current
            current = current.next
            count += 1
        
        item = current.data
        previous.next = current.next
        return item
    
    def display(self):
        """显示链表内容"""
        elements = []
        current = self.head
        while current is not None:
            elements.append(str(current.data))
            current = current.next
        print(" -> ".join(elements) + " -> None")
```

## 两种链表的比较

| 操作 | 无序链表 | 有序链表 |
|------|----------|----------|
| 添加元素 | O(1) - 添加到头部 | O(n) - 需要找到正确位置 |
| 查找元素 | O(n) - 需要遍历整个链表 | O(n) - 但可以提前终止 |
| 删除元素 | O(n) - 需要找到元素 | O(n) - 但可以提前终止 |
| 获取大小 | O(n) - 需要遍历计数 | O(n) - 需要遍历计数 |

## 应用场景

### 无序链表适用场景：
- 需要频繁在头部插入元素的场景
- 元素顺序不重要的场景
- 实现栈数据结构

### 有序链表适用场景：
- 需要保持元素有序的场景
- 需要频繁查找但插入删除较少的场景
- 实现优先队列

这两种链表实现展示了链表的基本操作，并体现了有序和无序数据结构在设计上的差异。
