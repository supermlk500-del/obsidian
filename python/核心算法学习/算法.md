### 1. 排序算法（重点掌握2个）
**快速排序（竞赛首选）**
```python
def quick_sort(arr, l, r):
    if l >= r: return
    i, j, pivot = l, r, arr[(l + r) // 2]
    while i <= j:
        while arr[i] < pivot: i += 1
        while arr[j] > pivot: j -= 1
        if i <= j:
            arr[i], arr[j] = arr[j], arr[i]
            i += 1
            j -= 1
    quick_sort(arr, l, j)
    quick_sort(arr, i, r)
```

**归并排序（稳定排序）**
```python
def merge_sort(arr, l, r):
    if l >= r: return
    mid = (l + r) // 2
    merge_sort(arr, l, mid)
    merge_sort(arr, mid + 1, r)
    
    # 合并
    tmp, i, j = [], l, mid + 1
    while i <= mid and j <= r:
        if arr[i] <= arr[j]:
            tmp.append(arr[i])
            i += 1
        else:
            tmp.append(arr[j])
            j += 1
    tmp.extend(arr[i:mid + 1])
    tmp.extend(arr[j:r + 1])
    arr[l:r + 1] = tmp
```

### 2. 查找算法
**二分查找（万能模板）**
```python
def binary_search(arr, target):
    l, r = 0, len(arr) - 1
    while l <= r:
        mid = (l + r) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            l = mid + 1
        else:
            r = mid - 1
    return -1

# 找第一个>=target的位置
def lower_bound(arr, target):
    l, r = 0, len(arr) - 1
    while l <= r:
        mid = (l + r) // 2
        if arr[mid] >= target:
            r = mid - 1
        else:
            l = mid + 1
    return l
```

### 3. 回溯算法（排列组合模板）
**全排列**
```python
def permute(nums):
    def backtrack(path):
        if len(path) == len(nums):
            res.append(path[:])
            return
        for num in nums:
            if num not in path:
                path.append(num)
                backtrack(path)
                path.pop()
    res = []
    backtrack([])
    return res
```

**组合**
```python
def combine(n, k):
    def backtrack(start, path):
        if len(path) == k:
            res.append(path[:])
            return
        for i in range(start, n + 1):
            path.append(i)
            backtrack(i + 1, path)
            path.pop()
    res = []
    backtrack(1, [])
    return res
```

### 4. 动态规划（基础模板）
**斐波那契DP**
```python
def fib(n):
    if n <= 1: return n
    dp = [0] * (n + 1)
    dp[1] = 1
    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]
```

**01背包**
```python
def knapsack(weights, values, capacity):
    n = len(weights)
    dp = [0] * (capacity + 1)
    for i in range(n):
        for w in range(capacity, weights[i] - 1, -1):
            dp[w] = max(dp[w], dp[w - weights[i]] + values[i])
    return dp[capacity]
```

### 5. 图论搜索
**DFS**
```python
def dfs(graph, start):
    visited, stack = set(), [start]
    while stack:
        node = stack.pop()
        if node not in visited:
            visited.add(node)
            stack.extend(neighbor for neighbor in graph[node] if neighbor not in visited)
    return visited
```

**BFS**
```python
from collections import deque

def bfs(graph, start):
    visited, queue = set(), deque([start])
    visited.add(start)
    while queue:
        node = queue.popleft()
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    return visited
```

### 1. 双指针
**快慢指针**
```python
def remove_duplicates(nums):
    if not nums: return 0
    slow = 0
    for fast in range(1, len(nums)):
        if nums[fast] != nums[slow]:
            slow += 1
            nums[slow] = nums[fast]
    return slow + 1
```

**左右指针**
```python
def two_sum(nums, target):
    left, right = 0, len(nums) - 1
    while left < right:
        total = nums[left] + nums[right]
        if total == target:
            return [left, right]
        elif total < target:
            left += 1
        else:
            right -= 1
    return [-1, -1]
```

### 2. 前缀和与差分
**前缀和**
```python
class PrefixSum:
    def __init__(self, nums):
        self.prefix = [0] * (len(nums) + 1)
        for i in range(len(nums)):
            self.prefix[i + 1] = self.prefix[i] + nums[i]
    
    def query(self, l, r):
        return self.prefix[r + 1] - self.prefix[l]
```

**差分数组**
```python
class Difference:
    def __init__(self, nums):
        self.diff = [0] * len(nums)
        self.diff[0] = nums[0]
        for i in range(1, len(nums)):
            self.diff[i] = nums[i] - nums[i - 1]
    
    def add_range(self, l, r, val):
        self.diff[l] += val
        if r + 1 < len(self.diff):
            self.diff[r + 1] -= val
    
    def get_nums(self):
        nums = [0] * len(self.diff)
        nums[0] = self.diff[0]
        for i in range(1, len(self.diff)):
            nums[i] = nums[i - 1] + self.diff[i]
        return nums
```

### 3. 滑动窗口
```python
def sliding_window(nums, k):
    from collections import defaultdict
    left = 0
    window = defaultdict(int)
    res = 0
    
    for right in range(len(nums)):
        window[nums[right]] += 1
        
        while len(window) > k:  # 窗口条件
            window[nums[left]] -= 1
            if window[nums[left]] == 0:
                del window[nums[left]]
            left += 1
        
        res = max(res, right - left + 1)
    return res
```

### 1. 质数判断

def is_prime(n):
    if n < 2: return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

# 埃氏筛
def sieve(n):
    is_prime = [True] * (n + 1)
    is_prime[0] = is_prime[1] = False
    for i in range(2, int(n**0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    return [i for i in range(2, n + 1) if is_prime[i]]
```

### 3. 快速幂

def fast_power(base, power, mod=None):
    result = 1
    while power > 0:
        if power & 1:
            result = result * base
            if mod: result %= mod
        base = base * base
        if mod: base %= mod
        power >>= 1
    return result
```

## 📊 四、数据结构模板

### 1. 并查集
```python
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        root_x, root_y = self.find(x), self.find(y)
        if root_x == root_y:
            return False
        if self.rank[root_x] < self.rank[root_y]:
            self.parent[root_x] = root_y
        else:
            self.parent[root_y] = root_x
            if self.rank[root_x] == self.rank[root_y]:
                self.rank[root_x] += 1
        return True
```
