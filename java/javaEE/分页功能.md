# 前端代码
```vue
<template>

    <div>

        <el-input v-model="input" style="width: 240px" placeholder="学生姓名" clearable />

        <el-button @click="btnClick" type="primary">查询</el-button>

    </div>

    <div v-if="stuList.length > 0">

        <el-table :data="stuList" style="width: 80%">

            <el-table-column label="编号" prop="id"></el-table-column>

            <el-table-column label="姓名" prop="stuName"></el-table-column>

            <el-table-column label="性别" prop="stuSex"></el-table-column>

            <el-table-column label="年龄" prop="stuAge"></el-table-column>

            <el-table-column label="手机号码" prop="stuPhone"></el-table-column>

        </el-table>

        <el-pagination background layout="prev, pager, next"

                    :total="page.total"

                    :default-page-size="page.size"

                    :default-current-page="page.current"

                    @prev-click="preClick"

                    @next-click="nextClick"

                    @current-change="changePage" />

    </div>

    <!-- 无数据显示"暂无数据" -->

    <el-empty v-else description="暂无数据~"></el-empty>

</template>

  

<script setup>

//导入axios

import axios from "@/api/request.js";

import { onMounted, reactive, ref } from "vue";

import { ElLoading, ElMessage, ElMessageBox, ElNotification } from "element-plus"; //导入加载框

  

const input = ref("")

const stuList = ref([])

const page = reactive({

    total: 0, //总条数

    pages: 0, //总页数

    size: 2,//每页显示的条数

    current:1 //当前页码

})

  

onMounted(() => {

    getData()

})

  

//点击上一页

const preClick = () => {

     page.current --

}

  

//点击下一页

const nextClick = () => {

    page.current ++

}

// 点击页码,index点击的页面

const changePage=(index)=>{

    page.current = index

    getData()

}

  

//点击查询按钮

const btnClick = ()=>{

    page.current = 1

    getData()

}

  

//获取数据

const getData = () => {

    openLoading() // 打开loading加载框

    const url = "/student/getByPage"

    const par = {

        stuName: input.value,

        currentPage: page.current, //要把页码和条数传递到后端

        pageSize: page.size

     }

    axios.get(url, { params: par }).then(res => {

        console.log("查询到的学生:", res.data)

        //查询到的学生数据

        stuList.value = res.data.data.records

        //分页数据

        page.total = res.data.data.total

        page.current = res.data.data.current

        page.size = res.data.data.size

        page.pages = res.data.data.pages

  

        loading.value.close()// 关闭loading加载框

        if (stuList.value.length > 0) {

            openMessage("success", "查询成功")

        } else {

            openMessage("info", "未查询到数据")

        }

  

    }).catch(err => {

        console.log("错误", err)

        loading.value.close()// 关闭loading加载框

        openNotification("error", "错误", "网络异常!")

    })

  

}

  

const loading = ref(false) //loading变量

//loadind加载框

const openLoading = () => {

    loading.value = ElLoading.service({

        lock: true,

        text: 'Loading',

        background: 'rgba(0, 0, 0, 0.7)',

    })

}

  

//通知弹出框

const openNotification = (type, title, content) => {

    ElNotification({

        type: type, // success|info|error|warning

        title: title,

        message: content,

        duration: 3000,  //存在的时间 ： 毫秒数   0表示不会自动消失

        position: "top-right",  // 通知显示的位置: top-left | top-right | bottom-left |..

        // offset: 100,  //偏移量

    })

}

  

//消息提示

const openMessage = (type, content) => {

    ElMessage({ //消息提示

        type: type, // success|info|error|warning

        message: content,

        // showClose:true //是否显示删除控件

    })

}

  
  

</script>

  

<style scoped></style>
```

# 后端代码

```java
@GetMapping("/getByPage")  
//请@RequestParam(defaultValue = "1") 请求参数默认值  
public ResponseData getByPage(@RequestParam(defaultValue = "1") Integer currentPage,  
                                @RequestParam(defaultValue = "1") Integer pageSize,Student stu){  
    QueryWrapper<Student> queryWrapper = new QueryWrapper<>();  
    //添加id查询条件  
    if (stu.getId() != null){  
        queryWrapper.eq("id", stu.getId());  
    }  
    if(stu.getStuName() != null){  
        queryWrapper.like("stu_name", stu.getStuName());  
    }  
    if(stu.getStuSex() != null){  
        queryWrapper.eq("stu_sex", stu.getStuSex());  
    }  
    if(stu.getStuAge() != null){  
        queryWrapper.eq("stu_age", stu.getStuAge());  
    }  
    if(stu.getStuPhone() != null){  
        queryWrapper.eq("stu_phone", stu.getStuPhone());  
    }  
  
    Page<Student> page = new Page<>(currentPage, pageSize);  
    IPage< Student> pageData = studentService.page(page, queryWrapper);  
    return new ResponseData("查询成功", pageData);  
}
```
