# main.ts
```ts
import { createApp } from 'vue' //导入vue依赖

import { createPinia } from 'pinia'//导入状态管理

import Antd from 'ant-design-vue'//淡如组件库

import 'ant-design-vue/dist/reset.css'//导入样式重置文件,统一浏览器默认样式

import '@/assets/styles/global.css'//@代表:src/. 这个文件一般用来放全局样式,全局字体等等

  

import App from './App.vue'//导入核心模块,整个网站的大容器

import router from './router'//导入路由系统

  

const app = createApp(App)//创建vue实例

  

app.use(createPinia())//给整个vue项目安装状态管理

app.use(Antd)//全局安装Ant Design Vue

app.use(router)//安装路由系统

  

app.mount('#app')//挂载,也最是把vue渲染到网页上
```

# ruoter
## index.ts
```ts
import { createRouter, createWebHistory } from 'vue-router'//导入vue-ruoter
//createWebHistory负责创建history路由样式

import { appRoutes } from '@/router/routes'//从routrs.ts导入路由表

  

const router = createRouter({//创建路由实例

  history: createWebHistory(import.meta.env.BASE_URL),//import.meta.env.BASE_URL 是Vite 的环境变量

  routes: [

    { path: '/', redirect: '/dashboard' },//用户访问"/"就自动跳转到"dashboard"

    ...appRoutes,//ES6展开运算符,把 appRoutes 里面所有路由展开

    { path: '/:pathMatch(.*)*',redirect: '/dashboard',},//用户访问乱七八糟的路径会自动跳转到dashboard./:pathMatch(.*)*是匹配所有未知路径

  ],

})

  

export default router//导出然后再main.ts中被安装也就是app.use(ruoter)
```
